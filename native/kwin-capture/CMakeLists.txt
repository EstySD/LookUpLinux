cmake_minimum_required(VERSION 3.22)
project(lookupcapture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin 6.8 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus Widgets OpenGL)
find_package(KF6Config REQUIRED)
add_library(lookupcapture MODULE capture.cpp)
target_link_libraries(lookupcapture PRIVATE KWin::kwin Qt6::DBus KF6::ConfigCore)
set_target_properties(lookupcapture PROPERTIES PREFIX "")
install(TARGETS lookupcapture LIBRARY DESTINATION lib/qt6/plugins/kwin/plugins)
install(FILES CMakeLists.txt capture.cpp metadata.json README.md DESTINATION share/lookup/kwin-capture)
